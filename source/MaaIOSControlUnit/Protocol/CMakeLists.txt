# Shared implementation for MaaToolkit and MaaIOSControlUnit.
set(maa_ios_protocol_root "${CMAKE_CURRENT_SOURCE_DIR}")
function(maa_ios_apply_protocol_options target visibility)
    target_include_directories(${target} ${visibility} "${maa_ios_protocol_root}")
    target_include_directories(${target} SYSTEM PRIVATE "${mdns_SOURCE_DIR}")
    target_compile_features(${target} ${visibility} cxx_std_20)
    if(MAA_IOS_LIBAV)
        target_compile_definitions(${target} ${visibility} MAA_IOS_HAVE_LIBAV=1)
    endif()
    target_compile_definitions(${target} PRIVATE OPENSSL_SUPPRESS_DEPRECATED)
    target_link_libraries(${target} ${visibility}
        MaaUtils ${OpenCV_LIBS} OpenSSL::SSL OpenSSL::Crypto MaaIOSLibAv Threads::Threads
        pugixml::pugixml MaaIOSLwip MaaIOSCurl)
    if(APPLE)
        target_link_libraries(${target} ${visibility} "-framework VideoToolbox" "-framework CoreMedia"
            "-framework CoreVideo" "-framework CoreFoundation" "-framework IOKit")
    elseif(WIN32)
        target_compile_definitions(${target} ${visibility} WIN32_LEAN_AND_MEAN NOMINMAX)
        if(MSVC)
            target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
        endif()
        target_link_libraries(${target} ${visibility} ws2_32 advapi32 iphlpapi)
    endif()
endfunction()

set(maa_ios_stop_token_check "#include <stop_token>\n#include <thread>\nint main(){ std::stop_source s; std::stop_callback c(s.get_token(), []{}); std::jthread t([](std::stop_token){}); t.request_stop(); t.join(); }")
check_cxx_source_compiles("${maa_ios_stop_token_check}" MAA_IOS_STANDARD_STOP_TOKEN)
if(NOT MAA_IOS_STANDARD_STOP_TOKEN AND CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    set(CMAKE_REQUIRED_FLAGS "-fexperimental-library")
    check_cxx_source_compiles("${maa_ios_stop_token_check}" MAA_IOS_EXPERIMENTAL_STOP_TOKEN)
    if(MAA_IOS_EXPERIMENTAL_STOP_TOKEN)
        target_compile_options(MaaIOSProtocol PUBLIC -fexperimental-library)
        target_link_options(MaaIOSProtocol PUBLIC -fexperimental-library)
    endif()
endif()
cmake_pop_check_state()
