cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(MaaIOSProtocol LANGUAGES C CXX)
    set(CMAKE_CXX_STANDARD 20)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
endif()

include(FetchContent)
if(POLICY CMP0135)
    cmake_policy(SET CMP0135 NEW)
endif()
option(MAA_IOS_FETCH_DEPENDENCIES "Fetch missing pinned iOS protocol dependencies" ON)
option(MAA_IOS_PROTOCOL_TESTS "Build iOS protocol offline tests" OFF)

find_package(nlohmann_json 3.12 QUIET CONFIG)
if(NOT TARGET nlohmann_json::nlohmann_json)
    if(NOT MAA_IOS_FETCH_DEPENDENCIES AND NOT FETCHCONTENT_SOURCE_DIR_NLOHMANN_JSON)
        message(FATAL_ERROR "MaaIOSProtocol requires nlohmann_json >= 3.12")
    endif()
    set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(nlohmann_json
        URL https://codeload.github.com/nlohmann/json/tar.gz/refs/tags/v3.12.0
        URL_HASH SHA256=4b92eb0c06d10683f7447ce9406cb97cd4b453be18d7279320f7b2f025c10187)
    FetchContent_MakeAvailable(nlohmann_json)
endif()
find_package(pugixml 1.14 QUIET CONFIG)
if(NOT TARGET pugixml::pugixml)
    if(NOT MAA_IOS_FETCH_DEPENDENCIES AND NOT FETCHCONTENT_SOURCE_DIR_PUGIXML)
        message(FATAL_ERROR "MaaIOSProtocol requires pugixml >= 1.14")
    endif()
    set(PUGIXML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(pugixml
        URL https://codeload.github.com/zeux/pugixml/tar.gz/refs/tags/v1.16
        URL_HASH SHA256=357bcab8877dc9943f355d3a72daba1b053238ba955f50fa81586afb65090219)
    FetchContent_MakeAvailable(pugixml)
endif()

# lwIP's build-time configuration determines its ABI, so build the pinned sources.
enable_language(C)
if(NOT MAA_IOS_FETCH_DEPENDENCIES AND NOT FETCHCONTENT_SOURCE_DIR_LWIP)
    message(FATAL_ERROR "Offline iOS builds require FETCHCONTENT_SOURCE_DIR_LWIP")
endif()
FetchContent_Declare(lwip
    URL https://codeload.github.com/lwip-tcpip/lwip/tar.gz/refs/tags/STABLE-2_2_1_RELEASE
    URL_HASH SHA256=ce0b7461c0ad9602c376f0bf07c5eb7253b48c7bf66f011c6bf3e2a96731c539
    SOURCE_SUBDIR maa-ios-core-only)
FetchContent_MakeAvailable(lwip)
file(GLOB maa_ios_lwip_core CONFIGURE_DEPENDS "${lwip_SOURCE_DIR}/src/core/*.c" "${lwip_SOURCE_DIR}/src/core/ipv6/*.c")
add_library(MaaIOSLwip STATIC ${maa_ios_lwip_core} net/LwipPort.cpp)
target_include_directories(MaaIOSLwip PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/net/lwip" "${lwip_SOURCE_DIR}/src/include")
set_target_properties(MaaIOSLwip PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON POSITION_INDEPENDENT_CODE ON)
# Upstream conditional IPv6 code leaves parameters and timer state unused in our selected
# configuration. Keep the framework's own warning policy intact.
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(MaaIOSLwip PRIVATE -Wno-error=unused-parameter -Wno-error=unused-variable)
endif()
if(CMAKE_C_COMPILER_ID MATCHES "Clang")
    target_compile_options(MaaIOSLwip PRIVATE -Wno-error=tautological-constant-out-of-range-compare)
endif()

if(NOT MAA_IOS_FETCH_DEPENDENCIES AND NOT FETCHCONTENT_SOURCE_DIR_MDNS)
    message(FATAL_ERROR "Offline iOS builds require FETCHCONTENT_SOURCE_DIR_MDNS")
endif()
FetchContent_Declare(mdns
    URL https://codeload.github.com/mjansson/mdns/tar.gz/refs/tags/1.4.3
    URL_HASH SHA256=be1fd8e35599cb7de179decbd0633c121d11a2dcb9cc193ff5c590bd0d480483
    SOURCE_SUBDIR maa-ios-header-only)
FetchContent_MakeAvailable(mdns)

find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
find_package(PkgConfig QUIET)
if(PkgConfig_FOUND)
    pkg_check_modules(MAA_IOS_LIBAV QUIET IMPORTED_TARGET libavcodec libavutil libswscale)
endif()
add_library(MaaIOSLibAv INTERFACE)
if(TARGET PkgConfig::MAA_IOS_LIBAV)
    target_link_libraries(MaaIOSLibAv INTERFACE PkgConfig::MAA_IOS_LIBAV)
else()
    # vcpkg / native MSVC packages expose ordinary headers and import libraries.
    find_path(MAA_IOS_FFMPEG_INCLUDE_DIR libavcodec/avcodec.h REQUIRED)
    find_library(MAA_IOS_AVCODEC_LIBRARY NAMES avcodec REQUIRED)
    find_library(MAA_IOS_AVUTIL_LIBRARY NAMES avutil REQUIRED)
    find_library(MAA_IOS_SWSCALE_LIBRARY NAMES swscale REQUIRED)
    target_include_directories(MaaIOSLibAv SYSTEM INTERFACE "${MAA_IOS_FFMPEG_INCLUDE_DIR}")
    target_link_libraries(MaaIOSLibAv INTERFACE "${MAA_IOS_AVCODEC_LIBRARY}"
        "${MAA_IOS_AVUTIL_LIBRARY}" "${MAA_IOS_SWSCALE_LIBRARY}")
endif()

add_library(MaaIOSCurl INTERFACE)
if(APPLE)
    if(IS_ABSOLUTE "${CMAKE_OSX_SYSROOT}")
        set(maa_ios_sdk "${CMAKE_OSX_SYSROOT}")
    else()
        set(maa_ios_sdk_name "${CMAKE_OSX_SYSROOT}")
        if(NOT maa_ios_sdk_name)
            set(maa_ios_sdk_name macosx)
        endif()
        execute_process(COMMAND xcrun --sdk "${maa_ios_sdk_name}" --show-sdk-path
            OUTPUT_VARIABLE maa_ios_sdk OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
    endif()
    target_link_libraries(MaaIOSCurl INTERFACE "${maa_ios_sdk}/usr/lib/libcurl.tbd")
else()
    find_package(CURL REQUIRED)
    if(PkgConfig_FOUND)
        pkg_check_modules(MAA_IOS_CURL QUIET IMPORTED_TARGET libcurl)
    endif()
    if(TARGET PkgConfig::MAA_IOS_CURL)
        target_link_libraries(MaaIOSCurl INTERFACE PkgConfig::MAA_IOS_CURL)
        foreach(maa_ios_curl_library IN LISTS MAA_IOS_CURL_LINK_LIBRARIES)
            if(maa_ios_curl_library MATCHES "(^|[/\\])libcurl\\.a$")
                target_link_directories(MaaIOSCurl INTERFACE ${MAA_IOS_CURL_STATIC_LIBRARY_DIRS})
                target_link_libraries(MaaIOSCurl INTERFACE ${MAA_IOS_CURL_STATIC_LIBRARIES})
                target_link_options(MaaIOSCurl INTERFACE ${MAA_IOS_CURL_STATIC_LDFLAGS_OTHER})
                break()
            endif()
        endforeach()
    else()
        target_link_libraries(MaaIOSCurl INTERFACE CURL::libcurl)
    endif()
endif()

file(GLOB_RECURSE maa_ios_protocol_sources CONFIGURE_DEPENDS *.cpp *.h)
list(FILTER maa_ios_protocol_sources EXCLUDE REGEX "/(tests|CMakeFiles)/")
list(FILTER maa_ios_protocol_sources EXCLUDE REGEX "/net/LwipPort\\.cpp$")
if(APPLE)
    list(FILTER maa_ios_protocol_sources EXCLUDE REGEX "/decode/PlatformDecoderFallback\\.cpp$")
else()
    list(FILTER maa_ios_protocol_sources EXCLUDE REGEX "/decode/VideoToolboxDecoder\\.cpp$")
endif()
if(NOT WIN32)
    list(FILTER maa_ios_protocol_sources EXCLUDE REGEX "/wifi/PairRecordWindows\\.(cpp|h)$")
endif()
add_library(MaaIOSProtocol STATIC ${maa_ios_protocol_sources})
set_target_properties(MaaIOSProtocol PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(MaaIOSProtocol PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_include_directories(MaaIOSProtocol SYSTEM PRIVATE "${mdns_SOURCE_DIR}")
target_compile_features(MaaIOSProtocol PUBLIC cxx_std_20)
target_compile_definitions(MaaIOSProtocol PUBLIC MAA_IOS_HAVE_LIBAV=1 PRIVATE OPENSSL_SUPPRESS_DEPRECATED)
target_link_libraries(MaaIOSProtocol PUBLIC OpenSSL::SSL OpenSSL::Crypto nlohmann_json::nlohmann_json MaaIOSLibAv Threads::Threads
    PRIVATE pugixml::pugixml MaaIOSLwip MaaIOSCurl)
if(APPLE)
    target_link_libraries(MaaIOSProtocol PRIVATE "-framework VideoToolbox" "-framework CoreMedia"
        "-framework CoreVideo" "-framework CoreFoundation" "-framework IOKit")
elseif(WIN32)
    target_compile_definitions(MaaIOSProtocol PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(MaaIOSProtocol PUBLIC ws2_32 PRIVATE advapi32 iphlpapi)
endif()

include(CheckCXXSourceCompiles)
include(CMakePushCheckState)
cmake_push_check_state(RESET)
set(CMAKE_REQUIRED_LIBRARIES Threads::Threads)
set(maa_ios_stop_token_check "#include <stop_token>\n#include <thread>\nint main(){ std::stop_source s; std::stop_callback c(s.get_token(), []{}); std::jthread t([](std::stop_token){}); t.request_stop(); t.join(); }")
check_cxx_source_compiles("${maa_ios_stop_token_check}" MAA_IOS_STANDARD_STOP_TOKEN)
if(NOT MAA_IOS_STANDARD_STOP_TOKEN AND CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    set(CMAKE_REQUIRED_FLAGS "-fexperimental-library")
    check_cxx_source_compiles("${maa_ios_stop_token_check}" MAA_IOS_EXPERIMENTAL_STOP_TOKEN)
    if(MAA_IOS_EXPERIMENTAL_STOP_TOKEN)
        target_compile_options(MaaIOSProtocol PUBLIC -fexperimental-library)
        target_link_options(MaaIOSProtocol PUBLIC -fexperimental-library)
    endif()
endif()
cmake_pop_check_state()
if(NOT MAA_IOS_STANDARD_STOP_TOKEN AND NOT MAA_IOS_EXPERIMENTAL_STOP_TOKEN)
    message(FATAL_ERROR "MaaIOSProtocol requires std::stop_token and std::jthread")
endif()

if(MAA_IOS_PROTOCOL_TESTS)
    enable_testing()
    add_subdirectory(tests)
endif()
