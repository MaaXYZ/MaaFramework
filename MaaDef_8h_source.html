145enum MaaInferenceExecutionProviderEnum
146{
147 // I don't recommend setting up MaaResOption_InferenceDevice in this case,
148 // because you don't know which EP will be used on different user devices.
149 // Auto tries CUDA, then DirectML, then WebGPU, then CoreML, and falls back to CPU.
150 MaaInferenceExecutionProvider_Auto = 0,
151
152 // MaaResOption_InferenceDevice will not work.
153 MaaInferenceExecutionProvider_CPU = 1,
154
155 // MaaResOption_InferenceDevice will be used to set adapter id,
156 // It's from Win32 API `EnumAdapters1`.
157 MaaInferenceExecutionProvider_DirectML = 2,
158
159 // Deprecated. MaaDeps no longer ships the CoreML execution provider. Use WebGPU.
160 // The value is kept so existing callers still compile. MaaResOption_InferenceDevice
161 // used to be a coreml_flag from
162 // https://github.com/microsoft/onnxruntime/blob/main/include/onnxruntime/core/providers/coreml/coreml_provider_factory.h
163 // [[deprecated]] is C++ only. __declspec(deprecated) cannot follow an enumerator name on MSVC.
164#if defined(__cplusplus)
165 MaaInferenceExecutionProvider_CoreML [[deprecated]] = 3,
166#else
167 MaaInferenceExecutionProvider_CoreML = 3,
168#endif
169
170 // MaaResOption_InferenceDevice will be used to set NVIDIA GPU ID
171 // TODO!
172 MaaInferenceExecutionProvider_CUDA = 4,
173
174 // MaaResOption_InferenceDevice is the WebGPU device id
175 // (ep.webgpuexecutionprovider.deviceId). Auto uses device 0.
176 // Shipped on Windows, Linux and macOS.
177 MaaInferenceExecutionProvider_WebGPU = 5,
178
179 // NCNN, only available when MaaFW is built with NCNN (Android only).
180 // Setting it on other builds fails (MaaResourceSetOption returns false).
181 // MaaResOption_InferenceDevice: CPU / Auto -> NCNN CPU; GPU is not supported yet, other values fail the loading.
182 // Not tried by Auto.
183 // Models are loaded from `<model stem>.ncnn.param/.ncnn.bin/.ncnn.json` (the "model" field with .onnx replaced),
184 // the .onnx file itself is not needed. There is no fallback to onnxruntime: a missing or broken ncnn model fails
185 // the recognition, so ship ncnn models together with this provider.
186 // OCR also runs on NCNN: `det.ncnn.param/.bin`, `rec.ncnn.param/.bin` and `keys.txt` in the OCR model dir, no fallback either.
187 // Runs in fp32 by default. fp16 (use_fp16_packed / use_fp16_storage / use_fp16_arithmetic) can be enabled per model in the "options" of the .ncnn.json
188 // (optional `det.ncnn.json` / `rec.ncnn.json` for OCR), see docs.
189 MaaInferenceExecutionProvider_NCNN = 6,
190};
