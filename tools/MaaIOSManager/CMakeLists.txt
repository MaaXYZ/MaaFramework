cmake_minimum_required(VERSION 3.28)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(MaaIOSManagerStandalone LANGUAGES C CXX)
    set(CMAKE_CXX_STANDARD 20)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
    add_subdirectory(../../source/MaaIOSControlUnit/Protocol protocol)
endif()

add_executable(MaaIOSManager main.cpp Cli.cpp Commands.cpp Prepare.cpp)
target_link_libraries(MaaIOSManager PRIVATE MaaIOSProtocol)
target_compile_features(MaaIOSManager PRIVATE cxx_std_20)
set_target_properties(MaaIOSManager PROPERTIES FOLDER Tools)
install(TARGETS MaaIOSManager RUNTIME DESTINATION bin)

option(BUILD_IOS_MANAGER_TESTING "Build offline MaaIOSManager command tests" OFF)
if(BUILD_IOS_MANAGER_TESTING)
    enable_testing()
    add_executable(MaaIOSManagerCliTest tests/cli_test.cpp Cli.cpp)
    target_include_directories(MaaIOSManagerCliTest PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_compile_features(MaaIOSManagerCliTest PRIVATE cxx_std_20)
    add_test(NAME MaaIOSManagerCliTest COMMAND MaaIOSManagerCliTest)
    add_test(NAME MaaIOSManagerHelp COMMAND MaaIOSManager --help)
    add_test(NAME MaaIOSManagerInvalidOption COMMAND MaaIOSManager pair --json)
    set_tests_properties(MaaIOSManagerInvalidOption PROPERTIES WILL_FAIL TRUE)
endif()
