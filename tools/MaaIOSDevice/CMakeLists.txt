cmake_minimum_required(VERSION 3.28)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(MaaIOSDeviceStandalone LANGUAGES C CXX)
    set(CMAKE_CXX_STANDARD 20)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
    add_subdirectory(../../source/MaaIOSControlUnit/Protocol protocol)
endif()

add_executable(MaaIOSDevice main.cpp Cli.cpp Commands.cpp Prepare.cpp)
target_link_libraries(MaaIOSDevice PRIVATE MaaIOSProtocol)
target_compile_features(MaaIOSDevice PRIVATE cxx_std_20)
set_target_properties(MaaIOSDevice PROPERTIES FOLDER Tools)
install(TARGETS MaaIOSDevice RUNTIME DESTINATION bin)

option(BUILD_IOS_DEVICE_TESTING "Build offline MaaIOSDevice command tests" OFF)
if(BUILD_IOS_DEVICE_TESTING)
    enable_testing()
    add_executable(MaaIOSDeviceCliTest tests/cli_test.cpp Cli.cpp)
    target_include_directories(MaaIOSDeviceCliTest PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_compile_features(MaaIOSDeviceCliTest PRIVATE cxx_std_20)
    add_test(NAME MaaIOSDeviceCliTest COMMAND MaaIOSDeviceCliTest)
    add_test(NAME MaaIOSDeviceHelp COMMAND MaaIOSDevice --help)
    add_test(NAME MaaIOSDeviceInvalidOption COMMAND MaaIOSDevice pair --json)
    set_tests_properties(MaaIOSDeviceInvalidOption PROPERTIES WILL_FAIL TRUE)
endif()
